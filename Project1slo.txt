payment_mode VARCHAR(20),
salesperson VARCHAR(50)
);

21.Find the city with the highest sales.


Select city, Sum(total_amount) AS total_sales
from Store_sales
Group by city
order by  total_sales desc
limit 1;



Find the category generating the highest revenue.

Select Category, sum(total_amount) as highest_revenue
from store_sales
group by category
order by highest_revenue desc
limit 1;



Display monthly sales revenue.


Select sum(total_amount) as monthly_revenue,
year(Order_date) AS year,
month(order_date) AS month
from store_sales
group by year(order_date),month(order_date)
order by month;
